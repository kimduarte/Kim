function pvAtualizarSelectInstituicoes_(lista) { var ordenada = lista.filter(function (x) { return x; }).sort(); ['pvPainelInstituicao', 'pvfInstituicao'].forEach(function (id) { var sel = document.getElementById(id); var atual = sel.value; sel.innerHTML = '' + ordenada.map(function (i) { return ''; }).join(''); sel.value = atual; }); }
